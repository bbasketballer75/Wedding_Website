const prefetchAdjacentImages = (currentIndex: number, images: Photo[]) => {
  const toPrefetch = []
  if (currentIndex > 0) {
    toPrefetch.push(images[currentIndex - 1].url)
  }
  if (currentIndex < images.length - 1) {
    toPrefetch.push(images[currentIndex + 1].url)
  }
  toPrefetch.forEach(url => {
    const link = document.createElement('link')
    link.rel = 'prefetch'
    link.as = 'image'
    link.href = url
    document.head.appendChild(link)
  })
}
```

// Prefetch adjacent images for smooth lightbox navigation
const prefetchAdjacentImages = (currentIndex: number, images: Photo[]) => {
  const toPrefetch: string[] = []

  if (currentIndex > 0 && images[currentIndex - 1]) {
    toPrefetch.push(images[currentIndex - 1].url)
  }
  if (currentIndex < images.length - 1 && images[currentIndex + 1]) {
    toPrefetch.push(images[currentIndex + 1].url)
  }

  toPrefetch.forEach(url => {
    const link = document.createElement('link')
    link.rel = 'prefetch'
    link.as = 'image'
    link.href = url
    document.head.appendChild(link)
  })
}
```

Update nextImage action to call prefetch:
```typescript
nextImage: () => {
  const { selectedImageIndex, filteredImages } = get()
  if (selectedImageIndex !== null && selectedImageIndex < filteredImages.length - 1) {
    const newIndex = selectedImageIndex + 1
    set({ selectedImageIndex: newIndex })
    // Prefetch after navigation
    prefetchAdjacentImages(newIndex, filteredImages)
  }
},
```

Update previousImage action similarly:
```typescript
previousImage: () => {
  const { selectedImageIndex, filteredImages } = get()
  if (selectedImageIndex !== null && selectedImageIndex > 0) {
    const newIndex = selectedImageIndex - 1
    set({ selectedImageIndex: newIndex })
    // Prefetch after navigation
    prefetchAdjacentImages(newIndex, filteredImages)
  }
},
```

Note: filteredImages is GalleryImage[] not Photo[], so you may need to use a different field. If the store uses GalleryImage, extract url from GalleryImage (it has src field per types/index.ts). Adjust accordingly based on actual store type.
